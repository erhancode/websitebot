Static HTML integration

Use this when the site is not PHP. Copy the three tags below into the page (before </body>) and point apiChat at the deployed api/bot/chat.php.

<script>
  window.BOT_WIDGET_CONFIG = {
    brandName: "Your Company",
    assistantName: "Ava",
    subtitle: "AI Assistant",
    avatar: "/assets/img/bot-avatar.svg",
    primaryColor: "#004d40",
    accentColor: "#c6a15b",
    apiChat: "/api/bot/chat.php",
    lang: document.documentElement.lang || "en",
    position: "left"
  };
</script>
<link rel="stylesheet" href="/assets/css/bot-widget.css">
<script src="/assets/js/bot-widget.js" defer></script>
